## 4. Service Worker Architecture

Using **Workbox** (via `next-pwa`) with a custom service worker layer.

### 4.1 Service Worker Registration

```typescript
// app/layout.tsx — register SW
if ('serviceWorker' in navigator) {
  window.addEventListener('load', () => {
    navigator.serviceWorker.register('/sw.js').then(reg => {
      console.log('SW registered:', reg.scope);
    });
  });
}
```

### 4.2 Caching Strategies

```javascript
// sw.js (generated by next-pwa + custom additions)
import { precacheAndRoute, cleanupOutdatedCaches } from 'workbox-precaching';
import { registerRoute } from 'workbox-routing';
import { CacheFirst, NetworkFirst, StaleWhileRevalidate } from 'workbox-strategies';
import { ExpirationPlugin } from 'workbox-expiration';
import { BackgroundSyncPlugin } from 'workbox-background-sync';

// Precache all Next.js build output
precacheAndRoute(self.__WB_MANIFEST);
cleanupOutdatedCaches();

// Strategy 1: CacheFirst — static assets (JS, CSS, fonts, icons)
registerRoute(
  ({ request }) => ['style', 'script', 'worker', 'font'].includes(request.destination),
  new CacheFirst({
    cacheName: 'static-assets',
    plugins: [
      new ExpirationPlugin({ maxAgeSeconds: 30 * 24 * 60 * 60, maxEntries: 200 })
    ]
  })
);

// Strategy 2: CacheFirst — images
registerRoute(
  ({ request }) => request.destination === 'image',
  new CacheFirst({
    cacheName: 'images',
    plugins: [
      new ExpirationPlugin({ maxAgeSeconds: 7 * 24 * 60 * 60, maxEntries: 100 })
    ]
  })
);

// Strategy 3: NetworkFirst — Supabase API calls
registerRoute(
  ({ url }) => url.hostname.includes('supabase.co'),
  new NetworkFirst({
    cacheName: 'supabase-api',
    networkTimeoutSeconds: 5,
    plugins: [
      new ExpirationPlugin({ maxAgeSeconds: 5 * 60, maxEntries: 100 })
    ]
  })
);

// Strategy 4: StaleWhileRevalidate — Next.js pages
registerRoute(
  ({ request }) => request.mode === 'navigate',
  new StaleWhileRevalidate({
    cacheName: 'pages',
    plugins: [
      new ExpirationPlugin({ maxAgeSeconds: 24 * 60 * 60 })
    ]
  })
);

// Strategy 5: NetworkOnly with BackgroundSync — write operations
const bgSyncPlugin = new BackgroundSyncPlugin('write-queue', {
  maxRetentionTime: 24 * 60  // Retry for 24 hours
});

registerRoute(
  ({ url, request }) => url.hostname.includes('supabase.co') && 
                        ['POST', 'PATCH', 'DELETE'].includes(request.method),
  new NetworkFirst({
    plugins: [bgSyncPlugin]
  }),
  'POST'
);
```

### 4.3 Offline Fallback

## 5. IndexedDB Schema (Dexie.js)

```typescript
// lib/db.ts
import Dexie, { Table } from 'dexie';

export class GymFlowDB extends Dexie {
  members!: Table<Member>;
  memberships!: Table<Membership>;
  attendance!: Table<Attendance>;
  workoutPlans!: Table<WorkoutPlan>;
  workoutSessions!: Table<WorkoutSession>;
  measurements!: Table<Measurement>;
  notifications!: Table<Notification>;
  syncQueue!: Table<SyncQueueItem>;
  metadata!: Table<{ key: string; value: string; updated_at: string }>;

  constructor() {
    super('gymflow');

    this.version(1).stores({
      members: 'id, gym_id, updated_at',
      memberships: 'id, member_id, status, end_date',
      attendance: 'id, member_id, date',
      workoutPlans: 'id, member_id, is_active',
      workoutSessions: 'id, member_id, started_at',
      measurements: 'id, member_id, measured_at',
      notifications: 'id, member_id, sent_at, is_read',
      syncQueue: '++local_id, status, created_at, table_name',
      metadata: 'key'
    });
  }
}

export const db = new GymFlowDB();
```

### 5.1 Sync Queue Item Schema

```typescript
interface SyncQueueItem {
  local_id?: number;           // Auto-increment local ID
  table_name: string;          // e.g. 'workout_session_sets'
  operation: 'INSERT' | 'PATCH' | 'DELETE';
  payload: Record<string, unknown>;
  remote_id?: string;          // Set after successful sync
  status: 'pending' | 'syncing' | 'done' | 'error';
  error?: string;
  created_at: string;
  synced_at?: string;
}
```

### 5.2 Sync Queue Processor

```typescript
// lib/sync.ts
async function processSyncQueue() {
  if (!navigator.onLine) return;

  const pending = await db.syncQueue
    .where('status').equals('pending')
    .sortBy('created_at');

  for (const item of pending) {
    await db.syncQueue.update(item.local_id!, { status: 'syncing' });

    try {
      const { error } = await supabase
        .from(item.table_name)
        [item.operation === 'INSERT' ? 'insert' :
         item.operation === 'PATCH' ? 'update' : 'delete'](item.payload);

      if (error) throw error;

      await db.syncQueue.update(item.local_id!, {
        status: 'done',
        synced_at: new Date().toISOString()
      });
    } catch (err) {
      await db.syncQueue.update(item.local_id!, {
        status: 'error',
        error: String(err)
      });
    }
  }
}

// Register online event listener
window.addEventListener('online', processSyncQueue);
// Also process on app focus
document.addEventListener('visibilitychange', () => {
  if (document.visibilityState === 'visible') processSyncQueue();
});
```

## 6. Push Notifications

### 6.1 VAPID Key Setup

```bash
# Generate VAPID keys (one-time)
npx web-push generate-vapid-keys
```

Store:
- `NEXT_PUBLIC_VAPID_PUBLIC_KEY` — in client env
- `VAPID_PRIVATE_KEY` — in Supabase Edge Function secrets only

### 6.2 Subscription Flow

```typescript
// lib/push.ts
export async function subscribeToPush() {
  const registration = await navigator.serviceWorker.ready;

  const subscription = await registration.pushManager.subscribe({
    userVisibleOnly: true,
    applicationServerKey: urlBase64ToUint8Array(process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY!)
  });

  // Save subscription to Supabase
  await fetch('/functions/v1/register-push', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${session.access_token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      endpoint: subscription.endpoint,
      keys: {
        p256dh: arrayBufferToBase64(subscription.getKey('p256dh')),
        auth: arrayBufferToBase64(subscription.getKey('auth'))
      }
    })
  });
}
```

### 6.3 Push Event Handler (Service Worker)

```javascript
// sw.js
self.addEventListener('push', (event) => {
  if (!event.data) return;

  const data = event.data.json();

  event.waitUntil(
    self.registration.showNotification(data.title, {
      body: data.body,
      icon: '/icons/icon-192.png',
      badge: '/icons/badge-72.png',
      tag: data.tag || 'gymflow',
      data: { url: data.url || '/' },
      vibrate: [200, 100, 200],
      requireInteraction: data.requireInteraction || false,
      actions: data.actions || []
    })
  );
});

self.addEventListener('notificationclick', (event) => {
  event.notification.close();

  const url = event.notification.data?.url || '/';

  event.waitUntil(
    clients.matchAll({ type: 'window' }).then((clientList) => {
      for (const client of clientList) {
        if (client.url === url && 'focus' in client) return client.focus();
      }
      if (clients.openWindow) return clients.openWindow(url);
    })
  );
});
```

## 7. Install Flow

### 7.1 Install Prompt (Custom UI)

```typescript
// hooks/useInstallPrompt.ts
let deferredPrompt: BeforeInstallPromptEvent | null = null;

window.addEventListener('beforeinstallprompt', (e) => {
  e.preventDefault();
  deferredPrompt = e as BeforeInstallPromptEvent;
});

export function useInstallPrompt() {
  const [canInstall, setCanInstall] = useState(false);

  useEffect(() => {
    if (deferredPrompt) setCanInstall(true);
  }, []);

  async function install() {
    if (!deferredPrompt) return;
    deferredPrompt.prompt();
    const { outcome } = await deferredPrompt.userChoice;
    deferredPrompt = null;
    setCanInstall(false);
    return outcome; // 'accepted' | 'dismissed'
  }

  return { canInstall, install };
}
```

### 7.2 Show Conditions

Show the install banner when:
- `beforeinstallprompt` has fired (Android Chrome).
- User has visited ≥ 2 times (track in localStorage).
- Banner not dismissed ≥ 2 times.
- PWA not already installed (`window.matchMedia('(display-mode: standalone)').matches === false`).

### 7.3 iOS Manual Install Prompt

iOS Safari does not support `beforeinstallprompt`. Show a modal with instructions:

```
On iPhone/iPad:
1. Tap the Share button (□↑) in Safari
2. Scroll down and tap "Add to Home Screen"
3. Tap "Add"
```

Detect iOS: `navigator.userAgent.match(/iPhone|iPad|iPod/)`

## 10. Screen Wake Lock

Used on the Digital Membership Card screen.

```typescript
let wakeLock: WakeLockSentinel | null = null;

async function requestWakeLock() {
  if ('wakeLock' in navigator) {
    wakeLock = await navigator.wakeLock.request('screen');
  }
}

async function releaseWakeLock() {
  if (wakeLock) {
    await wakeLock.release();
    wakeLock = null;
  }
}
```

Reacquire on page visibility restore (iOS Safari releases wake lock on tab switch).
